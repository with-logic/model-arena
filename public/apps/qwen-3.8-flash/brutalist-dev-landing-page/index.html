[ 01 ]

Deploy

Push a repo, get a URL in every region. Rollback is a number.

import { Brut } from "@brut/sdk";

const brut = new Brut(process.env.BRUT_KEY);

const dpl = await brut.deploy({
  repo: "acme/api",
  target: "edge",
});

console.log(dpl.url);
// https://8f2a1c9e.brut.app
import os
from brut import Brut

brut = Brut(api_key=os.environ["BRUT_KEY"])

dpl = brut.deploy(
    repo="acme/api",
    target="edge",
)

print(dpl.url)
# https://8f2a1c9e.brut.app
curl -sS -X POST \
  https://api.brut.dev/v1/deploy \
  -H "authorization: bearer $BRUT_KEY" \
  -H "content-type: application/json" \
  -d '{"repo":"acme/api","target":"edge"}'

Observe

Tail logs as a stream. No dashboards to configure, nothing to click.

for await (const log of brut.logs.tail({
  service: "edge",
  level: "error",
})) {
  console.log(log.message);
}
for log in brut.logs.tail(
    service="edge",
    level="error",
):
    print(log.message)
curl -sS "https://api.brut.dev/v1/logs?\
service=edge&level=error&tail=1" \
  -H "authorization: bearer $BRUT_KEY"
