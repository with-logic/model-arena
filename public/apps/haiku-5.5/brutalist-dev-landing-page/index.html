01 / DeployPOST

One call ships it.

Send a repo and a region. Get a live URL, TLS and a rollback handle back in the same response.

deploy.<ext> 
const res = await fetch("https://api.slab.dev/v1/deployments", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.SLAB_KEY}`,
    "Content-Type": "application/json"
  },
  body: JSON.stringify({ repo: "acme/checkout", region: "eu-west-1" })
});

const { url } = await res.json();
console.log(url); // https://checkout.slab.app
import os, requests

res = requests.post(
    "https://api.slab.dev/v1/deployments",
    headers={"Authorization": f"Bearer {os.environ['SLAB_KEY']}"},
    json={"repo": "acme/checkout", "region": "eu-west-1"},
)
print(res.json()["url"])  # https://checkout.slab.app
curl -X POST https://api.slab.dev/v1/deployments \
  -H "Authorization: Bearer $SLAB_KEY" \
  -H "Content-Type: application/json" \
  -d '{"repo":"acme/checkout","region":"eu-west-1"}'
02 / ObserveGET · stream

Logs, live.

Follow any service as it runs. Lines arrive as they are written, with no polling and no agent to install.

logs.<ext> 
const res = await fetch("https://api.slab.dev/v1/logs?service=checkout&follow=true", {
  headers: { Authorization: `Bearer ${process.env.SLAB_KEY}` }
});

const reader = res.body.pipeThrough(new TextDecoderStream()).getReader();
for (;;) {
  const { value, done } = await reader.read();
  if (done) break;
  console.log(value);
}
import os, requests

with requests.get(
    "https://api.slab.dev/v1/logs",
    params={"service": "checkout", "follow": "true"},
    headers={"Authorization": f"Bearer {os.environ['SLAB_KEY']}"},
    stream=True,
) as res:
    for line in res.iter_lines(decode_unicode=True):
        print(line)
curl -N "https://api.slab.dev/v1/logs?service=checkout&follow=true" \
  -H "Authorization: Bearer $SLAB_KEY"
03 / SecurePOST

Rotate keys, no outage.

Issue a new key and keep the old one alive for a grace window. Nothing in production breaks at 3 a.m.

rotate.<ext> 
const res = await fetch("https://api.slab.dev/v1/keys/rotate", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.SLAB_KEY}`,
    "Content-Type": "application/json"
  },
  body: JSON.stringify({ grace: "15m" })
});

const { key } = await res.json();
console.log(key); // new key, old key valid for 15 minutes
import os, requests

res = requests.post(
    "https://api.slab.dev/v1/keys/rotate",
    headers={"Authorization": f"Bearer {os.environ['SLAB_KEY']}"},
    json={"grace": "15m"},
)
print(res.json()["key"])  # old key valid for 15 minutes
curl -X POST https://api.slab.dev/v1/keys/rotate \
  -H "Authorization: Bearer $SLAB_KEY" \
  -H "Content-Type: application/json" \
  -d '{"grace":"15m"}'
